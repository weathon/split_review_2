\title{Comparisons Are All You Need for Optimizing Smooth Functions}

\begin{abstract}
When optimizing machine learning models, there are various scenarios where gradient computations are challenging or even infeasible. Furthermore, in reinforcement learning (RL), preference-based RL that only compares between options has wide applications, including reinforcement learning with human feedback in large language models. In this paper, we systematically study optimization of a smooth function $f\colon\R^n\to\R$ only assuming an oracle that compares function values at two points and tells which is larger. When $f$ is convex, we give two algorithms using $\tilde{O}(n/\epsilon)$ and $\tilde{O}(n^{2})$ comparison queries to find an $\epsilon$-optimal solution, respectively. When $f$ is nonconvex, our algorithm uses $\tilde{O}(n/\epsilon^2)$ comparison queries to find an $\epsilon$-approximate stationary point. All these results match the best-known zeroth-order algorithms with function evaluation queries in $n$ dependence, thus suggest that \emph{comparisons are all you need for optimizing smooth functions using derivative-free methods}. In addition, we also give an algorithm for escaping saddle points and reaching an $\epsilon$-second order stationary point of a nonconvex $f$, using $\tilde{O}(n^{1.5}/\epsilon^{2.5})$ comparison queries.

\end{abstract}

\section{Introduction}




Optimization is pivotal in the realm of machine learning. For instance, advancements in stochastic gradient descent (SGD) such as ADAM~\cite{kingma2014adam}, Adagrad~\cite{duchi2011adaptive}, etc., serve as foundational methods for the training of deep neural networks. However, there exist scenarios where gradient computations are challenging or even infeasible, such as black-box adversarial attack on neural networks~\cite{papernot2017practical,madry2018towards,chen2017zoo} and policy search in reinforcement learning~\cite{salimans2017evolution,choromanski2018structured}. 
Consequently, zeroth-order optimization methods with function evaluations have gained prominence, with provable guarantee for convex optimization~\cite{duchi2015optimal,nesterov2017random} and nonconvex optimization~\cite{ghadimi2013stochastic,fang2018spider,jin2018local,ji2019improved,zhang2022zeroth,vlatakis2019efficiently,balasubramanian2022zeroth}.

Furthermore, optimization for machine learning has been recently soliciting for even less information. For instance, it is known that taking only signs of gradient descents still enjoy good performance~\cite{liu2018signsgd,li2023faster,bernstein2018signsgd}. Moreover, in the breakthrough of large language models (LLMs), reinforcement learning from human feedback (RLHF) played an important rule in training these LLMs, especially GPTs by OpenAI~\cite{ouyang2022training}. Compared to standard RL that applies function evaluation for rewards, RLHF is preference-based RL that only compares between options and tells which is better. There is emerging research interest in preference-based RL. Refs.~\cite{chen2022human,saha2023dueling,novoseller2020dueling,xu2020preference,zhu2023principled,tang2023zeroth} established provable guarantees for learning a near-optimal policy from preference feedback. Ref.~\cite{wang2023rlhf} proved that for a wide range of preference models, preference-based RL can be solved with small or no extra costs compared to those of standard reward-based RL. 

In this paper, we systematically study optimization of smooth functions using comparisons. Specifically, for a function $f\colon\R^n\to\R$, we define the \emph{comparison oracle} of $f$ as $O_{f}^{\comp}\colon\R^{n}\times\R^{n}\to\{-1,1\}$ such that
\begin{align}\label{eq:comparison}
O_{f}^{\comp}(\x,\y)=
\begin{cases}
1&\text{if $f(\x)\geq f(\y)$} \\
-1&\text{if $f(\x)\leq f(\y)$}
\end{cases}.
\end{align}
(When $f(\x)=f(\y)$, outputting either 1 or $-1$ is okay.)
We consider an $L$-smooth function $f\colon\R^n\to\R$, defined as
\begin{align*}
\|\nabla f(\x)-\nabla f(\y)\|\leq L\|\x-\y\|\quad\forall\,\x,\y\in\R^n.
\end{align*}
Furthermore, we say $f$ is \emph{$\rho$-Hessian Lipschitz} if 
\begin{align*}
\|\nabla^2 f(\x)-\nabla^2 f(\y)\|\leq \rho\|\x-\y\|\quad\forall\,\x,\y\in\R^n.
\end{align*}
In terms of the goal of optimization, we define:
\begin{itemize}[leftmargin=*]
\item $\x \in \R^n$ is an \emph{$\epsilon$-optimal point} if $f(\x) \leq f^* + \epsilon$, where $f^* \coloneqq \inf_{\x}f(\x)$.

\item $\x \in \R^n$ is an \emph{$\epsilon$-first-order stationary point ($\epsilon$-FOSP)} if $\|\nabla f(\x)\|\leq\epsilon$.

\item $\x \in \R^n$ is an \emph{$\epsilon$-second-order stationary point ($\epsilon$-SOSP)} if $\|\nabla f(\x)\|\leq\epsilon$ and $\lambda_{\min}(\nabla^2 f(\x))\geq-\sqrt{\rho\epsilon}$.\footnote{This is a standard definition among nonconvex optimization literature for escaping saddle points and reaching approximate second-order stationary points, see for instance~\cite{nesterov2006cubic,curtis2017trust,agarwal2017finding,carmon2018accelerated,jin2018accelerated,allen2018neon2,xu2018first,zhang2022zeroth,zhang2023faster}.}
\end{itemize}
\textbf{Our main results} can be listed as follows:
\begin{itemize}[leftmargin=*]
\item For an $L$-smooth convex $f$, \thm{CCO} finds an $\epsilon$-optimal point in $O(nL/\kern-0.2mm\epsilon\kern-0.3mm\log(nL/\epsilon))$ comparisons.

\item For an $L$-smooth convex $f$, \thm{cutting-plane}
finds an $\epsilon$-optimal point in $O(n^2\log(nL/\epsilon))$ comparisons.

\item For an $L$-smooth $f$, \thm{Comparison-FOSP}
finds an $\epsilon$-FOSP using $O(Ln\log n/\epsilon^2)$ comparisons.

\item For an $L$-smooth, $\rho$-Hessian Lipschitz $f$, \thm{Comparison-SOSP} finds an $\epsilon$-SOSP in $\tilde{O}(n^{1.5}/\epsilon^{2.5})$ comparisons.
\end{itemize}

Intuitively, our results can be described as \textbf{comparisons are all you need for derivative-free methods}: For finding an approximate minimum of a convex function, the state-of-the-art zeroth-order methods with full function evaluations have query complexities $O(n/\sqrt{\epsilon})$~\cite{nesterov2017random} or $\tilde{O}(n^{2})$~\cite{lee2017efficient}, which are matched in $n$ by our \thm{CCO} and \thm{cutting-plane} using comparisons, respectively. For finding an approximate stationary point of a nonconvex function, the state-of-the-art zeroth-order result has query complexity $O(n/\epsilon^{2})$~\cite{fang2018spider}, which is matched by our \thm{Comparison-FOSP} up to a logarithmic factor. 
In other words, in derivative-free scenarios for optimizing smooth functions, function values per se are unimportant but their comparisons, which indicate the direction that the function decreases. 

Among the literature for derivative-free optimization methods~\cite{larson2019derivative}, direct search methods~\cite{kolda2003optimization} proceed by comparing function values, including the directional direct search method~\cite{audet2006mesh} and the Nelder-Mead method~\cite{nelder1965simplex} as examples. However, the directional direct search method does not have a known rate of convergence, meanwhile the Nelson-Mead method may fail to converge to a stationary point for smooth functions~\cite{dennis1991direct}.
As far as we know, the most relevant result is by Bergou et al.~\cite{bergou2020stochastic}, which proposed the stochastic three points (STP) method and found an $\epsilon$-optimal point of a convex function and an $\epsilon$-FOSP of a nonconvex function in $\tilde{O}(n/\epsilon)$ and $\tilde{O}(n/\epsilon^{2})$ comparisons, respectively. STP also has a version with momentum~\cite{gorbunov2019stochastic}. Our \thm{CCO} and \thm{Comparison-FOSP} can be seen as rediscoveries of these results using different methods. However, for comparison-based convex optimization with $\poly(\log 1/\epsilon)$ dependence, Ref.~\cite{jamieson2012query} achieved this for strongly convex functions, and the state-of-the-art result for general convex optimization by Karabag et al.~\cite{karabag2021smooth} takes $\tilde{O}(n^{4})$ comparison queries. Their algorithm applies the ellipsoid method, which has $\tilde{O}(n^{2})$ iterations and each iteration takes $\tilde{O}(n^{2})$ comparisons to construct the ellipsoid. This $\tilde{O}(n^{4})$ bound is noticeably worse than our \thm{cutting-plane}. As far as we know, our \thm{Comparison-SOSP} is the \emph{first provable guarantee} for finding an $\epsilon$-SOSP of a nonconvex function by comparisons. 

\paragraph{Techniques.}
Our first technical contribution is \thm{Comparison-GDE}, which for a point $\x$ estimates the direction of $\nabla f(\x)$ within precision $\delta$. This is achieved by \algo{Comparison-GDE}, named as \texttt{Comparison-GDE} (GDE is the acronym for gradient direction estimation). It is built upon a directional preference subroutine (\algo{DP}), which inputs a unit vector $\v\in\R^{n}$ and a precision parameter $\Delta>0$, and outputs whether $\langle \grad f(\x), \v\rangle\geq-\Delta$ or $\langle \grad f(\x), \v\rangle\leq\Delta$ using the value of the comparison oracle for $O_{f}^{\comp}(\x+\frac{2\Delta}{L}\v,\x)$. 
\texttt{Comparison-GDE} then has three phases:
\begin{itemize}[leftmargin=*]
\item First, it sets $\v$ to be all standard basis directions $\vect{e}_i$ to determine the signs of all $\nabla_i f(\x)$ (up to $\Delta$). 

\item It then sets $\v$ as $\frac{1}{\sqrt{2}}(\e_i-\e_j)$, which can determine whether $|\nabla_i f(\x)|$ or $|\nabla_j f(\x)|$ is larger (up to $\Delta$). Start with $\e_1$ and $\e_2$ and keep iterating to find the $i^{*}$ with the largest $|\frac{\partial}{\partial i^{*}}\nabla f(\x)|$ (up to $\Delta$).

\item Finally, for each $i\neq i^{*}$, It then sets $\v$ to have form $\frac{1}{\sqrt{1+\alpha_{i}^{2}}}(\alpha_{i}\e_{i^{*}}-\e_i)$ and applies binary search to find the value for $\alpha_i$ such that $\alpha_i|\nabla_{i^*} f(\x)|$ equals to $|\nabla_i f(\x)|$ up to enough precision.
\end{itemize}
\texttt{Comparison-GDE} outputs $\boldsymbol{\alpha}/\|\boldsymbol{\alpha}\|$ for GDE, where $\alpha=(\alpha_1,\ldots,\alpha_n)^{\top}$. It in total uses $O(n\log(n/\delta))$ comparison queries, with the main cost coming from binary searches in the last step (the first two steps both take $\leq n$ comparisons).

We then leverage \texttt{Comparison-GDE} for solving various optimization problems. In convex optimization, we develop two algorithms that find an $\epsilon$-optimal point separately in \sec{NGD} and \sec{cutting}. Our first algorithm is a specialization of the adaptive version of normalized gradient descent (NGD) introduced in \cite{levy2017online}, where we replace the normalized gradient query in their algorithm by \texttt{Comparison-GDE}. It is a natural choice to apply gradient estimation to normalized gradient descent, given that the comparison model only allows us to estimate the gradient direction without providing information about its norm. Note that Ref.~\cite{bergou2020stochastic} also discussed NGD, but their algorithm using NGD still needs the full gradient and cannot be directly implemented by comparisons. Our second algorithm builds upon the framework of cutting plane methods, where we show that the output of \texttt{Comparison-GDE} is a valid separation oracle, as long as it is accurate enough. 

In nonconvex optimization, we develop two algorithms that  find an $\epsilon$-FOSP and an $\epsilon$-SOSP, respectively, in \sec{FOSP} and \sec{SOSP}. Our algorithm for finding an $\epsilon$-FOSP is a specialization of the NGD algorithm, where the normalized gradient is given by \texttt{Comparison-GDE}. Our algorithm for finding an $\epsilon$-SOSP uses a similar approach as corresponding first-order methods by \cite{allen2018neon2,xu2018first} and proceeds in rounds, where we alternately apply NGD and negative curvature descent to ensure that the function value will have a large decrease if more than $1/9$ of the iterations in this round are not $\epsilon$-SOSP. The normalized gradient descent part is essentially the same as our algorithm for $\epsilon$-FOSP in \sec{FOSP}. The negative curvature descent part with comparison information, however, is much more technically involved. In particular, previous first-order methods \cite{allen2018neon2,xu2018first, zhang2021escape} all contains a subroutine that can find a negative curvature direction near a saddle point $\x$ with $\lambda_{\min}(\nabla^2f(\x)\leq-\sqrt{\rho\epsilon})$. One crucial step in this subroutine is to approximate the Hessian-vector product $\nabla^2f(\x)\cdot\y$ for some unit vector $\y\in\R^n$ by taking the difference between $\nabla f(\x+r\y)$ and $\nabla f(\x)$, where $r$ is a very small parameter. However, this is infeasible in the comparison model which only allows us to estimate the gradient direction without providing information about its norm. Instead, we find the directions of $\nabla f(\x)$, $\nabla f(\x+r\y)$, and $\nabla f(\x-r\y)$ by \texttt{Comparison-GDE}, and we determine the direction of $\nabla f(\x+r\y)-f(\y)$ using the fact that its intersection with $\nabla f(\x)$ and $\nabla f(\x+r\y)$ as well as its intersection with $\nabla f(\x)$ and $\nabla f(\x-r\y)$ give two segments of same length (see \fig{triangle}).

\paragraph{Open questions.}
Our work leaves several natural directions for future investigation:
\begin{itemize}[leftmargin=*]
\item Can we give comparison-based optimization algorithms based on accelerated gradient descent (AGD) methods? This is challenging because AGD requires carefully chosen step sizes, but with comparisons we can only learn gradient directions but not the norm of gradients. This is also the main reason why the $1/\epsilon$ dependence in our \thm{CCO} and \thm{Comparison-SOSP} are worse than~\cite{nesterov2017random} and~\cite{zhang2023faster} with evaluations in their respective settings. 

\item Can we improve our result for finding second-order stationary points in nonconvex optimization? Compared to gradient-based methods that choose the step size in negative curvature finding~\cite{allen2018neon2,xu2018first}, our comparison-based perturbed normalized gradient descent (\algo{Comparison-PNGD}) can only utilize gradient directions but have no information about gradient norms, resulting in a fixed and conservative step size and in total $\tilde{O}(\sqrt{n}/\epsilon)$ iterations.

\item Can we apply our algorithms to machine learning? \cite{tang2023zeroth} made attempts on preference-based RL, and it is worth further exploring whether we can prove more theoretical results for preference-based RL and other machine learning settings. It would be also of general interest to see if our results can provide theoretical justification for quantization in neural networks~\cite{gholami2022survey}.
\end{itemize}

\paragraph{Notations.}
We use bold letters, e.g., $\x$, $\y$, to denote vectors and capital letters, e.g., $A$, $B$, to denote matrices. We use $\|\cdot\|$ to denote the Euclidean norm ($\ell_2$-norm) and denote $\mathcal{S}^{n-1}$ to be the $n$-dimensional sphere with radius 1, i.e., $\mathcal{S}^{n-1}:=\{\x\in\R^{n}:\|\x\|=1\}$. We denote $\B_R(\x) \coloneqq \{\y\in\R^n\colon\|\y-\x\|\leq R\}$ and $[T] \coloneqq \{0,1,\ldots,T\}$. 
For a convex set $\mathcal{K}\subseteq\R^n$, its diameter is defined as $D\coloneqq\sup_{\x,\y\in\mathcal{K}}\|\x-\y\|$ and its projection operator $\Pi_\mathcal{K}$ is defined as
\begin{align*}
\Pi_\mathcal{K}(\x)\coloneqq\mathrm{argmin}_{\y\in\mathcal{K}}\|\x-\y\|,\quad\forall\x\in\R^n.
\end{align*}




\section{Estimation of Gradient Direction by Comparisons}


First, we show that given a point $\x\in\R^{n}$ and a direction $\v\in\R^{n}$, we can use one comparison query to understand whether the inner product $\langle\grad f(\x),\v\rangle$ is roughly positive or negative. Intuitively, this inner product determines whether $\x+\v$ is following or against the direction of $\grad f(\x)$, also known as \emph{directional preference} (DP) in~\cite{karabag2021smooth}.

\begin{lemma}\label{lem:DP}
Given a point $\x\in\R^{n}$, a unit vector $\v\in\B_1(0)$, and precision $\Delta>0$ for directional preference. Then \algo{DP} is correct:
\begin{itemize}[leftmargin=*]
\item If $O_{f}^{\comp}(\x+\frac{2\Delta}{L}\v,\x)=1$, then $\langle \grad f(\x), \v\rangle\geq-\Delta$.
\item If $O_{f}^{\comp}(\x+\frac{2\Delta}{L}\v,\x)=-1$, then $\langle \grad f(\x), \v\rangle\leq\Delta$.
\end{itemize}
\end{lemma}

\begin{algorithm2e}
	\caption{DP(\x,\v,$\Delta$)}
	\label{algo:DP}
	\LinesNumbered
	\DontPrintSemicolon
    \KwInput{Comparison oracle $O_{f}^{\comp}$ of $f\colon\R^n\to\R$, $\x\in\R^n$, unit vector $\v\in\B_1(0)$, $\Delta>0$}
    \If{$O_{f}^{\comp}(\x+\frac{2\Delta}{L}\v,\x)=1$}{\Return ``$\langle \grad f(\x), \v\rangle\geq-\Delta$"}
    \Else({(in this case $O_{f}^{\comp}(\x+\frac{2\Delta}{L}\v,\x)=-1$)}){\Return ``$\langle \grad f(\x), \v\rangle\leq\Delta$"}
\end{algorithm2e}

\begin{proof}
Since $f$ is an $L$-smooth differentiable function,
\begin{align*}
|f(\y)-f(\x)-\langle\grad f(\x),\y-\x\rangle|\leq\frac{1}{2}L\|\y-\x\|^{2}
\end{align*}
for any $\x,\y\in\R^{n}$. Take $\y=\x+\frac{2\Delta}{L}\v$, this gives
\begin{align*}
\left|f(\y)-f(\x)-\frac{2\Delta}{L}\langle\grad f(\x),\v\rangle\right|\leq\frac{1}{2}L\left(\frac{2\Delta}{L}\right)^{2}=\frac{2\Delta^{2}}{L}.
\end{align*}
Therefore, if $O_{f}^{\comp}(\y,\x)=1$, i.e., $f(\y)\geq f(\x)$,
\begin{align*}
\frac{2\Delta}{L}\langle\grad f(\x),\v\rangle\kern-0.5mm\geq\kern-0.5mm \frac{2\Delta}{L}\langle\grad f(\x),\v\rangle+f(\x)-f(\y)\kern-0.5mm\geq\kern-0.5mm -\frac{2\Delta^{2}}{L}
\end{align*}
and hence $\langle\grad f(\x),\v\rangle\geq -\Delta$. On the other hand, if $O_{f}^{\comp}(\y,\x)=-1$, i.e., $f(\y)\leq f(\x)$,
\begin{align*}
\frac{2\Delta}{L}\langle\grad f(\x),\v\rangle\leq f(\y)-f(\x)+\frac{2\Delta^{2}}{L}\leq\frac{2\Delta^{2}}{L}
\end{align*}
and hence $\langle\grad f(\x),\v\rangle\leq\Delta$.
\end{proof}

Now, we prove that we can use $\tilde{O}(n)$ comparison queries to approximate the direction of the gradient at a point, which is one of our main technical contributions.

\begin{theorem}\label{thm:Comparison-GDE}
For an $L$-smooth function $f\colon\R^n\to\R$ and a point $\x\in\R^n$, \algo{Comparison-GDE} outputs an estimate $\tilde{\g}(\x)$ of the direction of $\nabla f(\x)$ using $O(n\log(n/\delta))$ queries to the comparison oracle $O_{f}^{\comp}$ of $f$ (Eq.~\eq{comparison}) that satisfies
\begin{align*}
\left\|\tilde{\g}(\x)-\frac{\nabla f(\x)}{\|\nabla f(\x)\|}\right\|\leq\delta
\end{align*}
if we are given a parameter $\gamma>0$ such that $\|\nabla f(\x)\|\geq \gamma$. 
\end{theorem}

\begin{algorithm2e}[!ht]
	\caption{Comparison-based Gradient Direction Estimation (Comparison-GDE($\x,\delta,\gamma$))}
	\label{algo:Comparison-GDE}
	\LinesNumbered
	\DontPrintSemicolon
    \KwInput{Comparison oracle $O_{f}^{\comp}$ of $f\colon\R^n\to\R$, precision $\delta$, lower bound $\gamma$ on $\|\grad f(\x)\|$}
    Set $\Delta\leftarrow \delta\gamma/4n^{3/2}$. Denote $\grad f(\x)=(g_1,\ldots,g_n)^{\top}$\;
    
    Call \algo{DP} with inputs $(\x,\e_1,\Delta),\ldots,(\x,\e_n,\Delta)$ where $e_i$ is the $i^{\text{th}}$ standard basis with $i^{\text{th}}$ coordinate being 1 and others being 0. This determines whether $g_i\geq-\Delta$ or $g_i\leq\Delta$ for each $i\in[n]$. WLOG 
    \begin{align}\label{eq:gi-sign}
    g_i\geq-\Delta\quad\forall i\in[n]
    \end{align}
    (otherwise take a minus sign for the $i^{\text{th}}$ coordinate)\label{lin:gi-sign}\;
    
    We next find the approximate largest one among $g_1,\ldots,g_n$. Call \algo{DP} with input $(\x,\frac{1}{\sqrt{2}}(\e_1-\e_2),\Delta)$. This determines whether $g_1\geq g_2-\sqrt{2}\Delta$ or $g_2\geq g_1-\sqrt{2}\Delta$. If the former, call \algo{DP} with input $(\x,\frac{1}{\sqrt{2}}(\e_1-\e_3),\Delta)$. If the later, call \algo{DP} with input $(\x,\frac{1}{\sqrt{2}}(\e_2-\e_3),\Delta)$. Iterate this until $e_n$, we find the $i^{*}\in[n]$ such that
    \begin{align}\label{eq:gi-largest}
    g_{i^*}\geq\max_{i\in[n]}g_i-\sqrt{2}\Delta
    \end{align}\label{lin:gi-largest}\;
    \For{$i=1$ \emph{to} $i=n$ \emph{(except $i=i^{*}$)}}
        {Initialize $\alpha_i\leftarrow 1/2$\;
        Apply binary search to $\alpha_i$ in $\lceil \log_2(\gamma/\Delta)+1\rceil$ iterations by calling \algo{DP} with input $(\x,\frac{1}{\sqrt{1+\alpha_{i}^{2}}}(\alpha_{i}\e_{i^{*}}-\e_i),\Delta)$. For the first iteration with $\alpha_i=1/2$, if $\alpha_i g_{i^*}-g_i\geq -\sqrt{2}\Delta$ we then take $\alpha_i=3/4$; if $\alpha_i g_{i^*}-g_i\leq\sqrt{2}\Delta$ we then take $\alpha_i=1/4$. Later iterations are similar. Upon finishing the binary search, $\alpha_i$ satisfies
        \begin{align}\label{eq:gi-value}
        g_i-\sqrt{2}\Delta\leq \alpha_i g_{i^*}\leq g_i+\sqrt{2}\Delta
        \end{align}\label{lin:gi-value}
        }
    \Return $\tilde{\g}(\x)=\frac{\boldsymbol{\alpha}}{\|\boldsymbol{\alpha}\|}$ where $\alpha=(\alpha_1,\ldots,\alpha_n)^{\top}$, $\alpha_i$ ($i\neq i^{*}$) is the output of the for loop, $\alpha_{i^{*}}=1$
\end{algorithm2e}

\begin{proof}
The correctness of \eq{gi-sign} and \eq{gi-largest} follows directly from the arguments in \lin{gi-sign} and \lin{gi-largest}, respectively. For \lin{gi-value}, since $\alpha_i\leq 1$ for any $i\in[n]$, the binary search can be regarded as having bins with interval lengths $\sqrt{1+\alpha_i^{2}}\Delta\leq\sqrt{2}\Delta$, and when the binary search ends Eq.~\eq{gi-value} is satisfied. Furthermore, Eq.~\eq{gi-value} can be written as
\begin{align*}
\left|\alpha_i-\frac{g_i}{g_{i^*}}\right|\leq\frac{\sqrt{2}\Delta}{g_{i^*}}\leq\frac{2\Delta\sqrt{n}}{\gamma}.
\end{align*}
This is because $\|\nabla f(\x)\|=\|(g_1,\ldots,g_n)^{\top}\|\geq \gamma$ implies $\max_{i\in[n]}g_i\geq\gamma/\sqrt{n}$, and together with \eq{gi-largest} we have $g_{i^*}\geq\gamma/\sqrt{n}-\sqrt{2}\Delta\geq \gamma/\sqrt{2n}$ because $\Delta\leq \gamma/4\sqrt{n}$.

We now estimate $\left\|\tilde{\g}(\x)-\frac{\nabla f(\x)}{\|\nabla f(\x)\|}\right\|$. Note $\frac{\nabla f(\x)}{\|\nabla f(\x)\|}=\frac{\nabla f(\x)/g_{i^*}}{\|\nabla f(\x)/g_{i^*}\|}$ and $\tilde{\g}(\x)=\boldsymbol{\alpha}/\|\boldsymbol{\alpha}\|$. Moreover 
\begin{align*}
\left\|\boldsymbol{\alpha}-\frac{\nabla f(\x)}{g_{i^*}}\right\|\leq\sum_{i=1}^{n}\left|\alpha_i-\frac{g_i}{g_{i^*}}\right|\leq \frac{2\Delta\sqrt{n}(n-1)}{\gamma}.
\end{align*}
By \lem{dist-norm-vector} for bounding distance between normalized vectors) and the fact that $\|\boldsymbol{\alpha}\|\geq 1$,
\begin{align*}
\left\|\tilde{\g}(\x)-\frac{\nabla f(\x)}{\|\nabla f(\x)\|}\right\|&=\left\|\frac{\boldsymbol{\alpha}}{\|\boldsymbol{\alpha}\|}-\frac{\nabla f(\x)/g_{i^*}}{\|\nabla f(\x)/g_{i^*}\|}\right\|\leq\frac{4\Delta n^{3/2}}{\gamma}\leq\delta.
\end{align*}
Thus the correctness has been established. For the query complexity, \lin{gi-sign} takes $n$ queries, \lin{gi-largest} takes $n-1$ queries, and \lin{gi-value} throughout the for loop takes $(n-1)\lceil \log_2(\gamma/\sqrt{2}\Delta)+1\rceil=O(n\log(n/\delta))$ queries to the comparison oracle, given that each $\alpha_i$ is within the range of $[0,1]$ and we approximate it to accuracy $\sqrt{2}\Delta/g_{i^*}\geq\sqrt{2}\Delta/\gamma$. This finishes the proof.
\end{proof}



\section{Convex Optimization by Comparisons}

In this section, we study convex optimization with function value comparisons:
\begin{problem}[Comparison-based convex optimization]\label{prob:CCO}
In the comparison-based convex optimization (CCO) problem we are given query access to a comparison oracle $O_{f}^{\comp}$ \eq{comparison} for an $L$-smooth convex function $f\colon\R^n\to\R$ whose minimum is achieved at $\x^*$ with $\|\x^*\|\leq R$. The goal is to output a point $\tilde{\x}$ such that $\|\tilde{\x}\|\leq R$ and $f(\tilde{\x})-f(\x^*)\leq\epsilon$, i.e., $\tilde{\x}$ is an $\epsilon$-optimal point.
\end{problem}
We provide two algorithms that solve \prob{CCO}. In \sec{NGD}, we use normalized gradient descent to achieve linear dependence in $n$ (up to a log factor) in terms of comparison queries. In \sec{cutting}, we use cutting plane method to achieve $\log(1/\epsilon)$ dependence in terms of comparison queries.

\subsection{Comparison-based adaptive normalized gradient descent}\label{sec:NGD}

In this subsection, we present our first algorithm for \prob{CCO}, \algo{Comparison-AdaNGD}, which applies \texttt{Comparison-GDE} (\algo{Comparison-GDE}) with estimated gradient direction at each iteration to the adaptive normalized gradient descent (AdaNGD), originally introduced by \cite{levy2017online}. 
\begin{algorithm2e}[ht]
	\caption{Comparison-based Approximate Adaptive Normalized Gradient Descent (Comparison-AdaNGD)}
	\label{algo:Comparison-AdaNGD}
	\LinesNumbered
	\DontPrintSemicolon
    \KwInput{Function $f\colon\R^n\to\R$, precision $\epsilon$, radius $R$}
    $T\leftarrow \frac{64LR^2}{\epsilon}$, $\delta\leftarrow\frac{1}{4R}\sqrt{\frac{\epsilon}{2L}}$, $\gamma\leftarrow\frac{\epsilon}{2R}$, $\x_0\leftarrow\0$\;
    \For{$t=0,\ldots,T-1$}{
        $\hat{\g}_t\leftarrow$\texttt{Comparison-GDE}$(\x_t,\delta,\gamma)$\label{lin:AdaNGD-GDE}\;
        $\eta_t\leftarrow R\sqrt{2/t}$ 
        \;
        $\x_{t+1}=\Pi_{\B_R(\0)}(\x_t-\eta_t\hat{\g}_t)$\;
    }
    $t_\out\leftarrow{\mathrm{argmin}}_{t\in[T]}f(\x_t)$\;
    \Return $\x_{t_\out}$
\end{algorithm2e}
\begin{theorem}\label{thm:CCO}
\algo{Comparison-AdaNGD} solves \prob{CCO} using $O(nLR^2/\epsilon\log(nLR^2/\epsilon))$ queries.
\end{theorem}

The following result bounds the rate at which \algo{Comparison-AdaNGD} decreases the function value of $f$.

\begin{lemma}\label{lem:Comparison-AdaNGD-convergence}
In the setting of \prob{CCO}, \algo{Comparison-AdaNGD} satisfies
\begin{align*}
\min_{t\in[T]}f(\x_t)-f^*\leq 2L(2R\sqrt{2T}+2T\delta R)^2/T^2,
\end{align*}
if at each step we have
\begin{align*}
\left\|\tilde{\g}_t-\frac{\nabla f_t(\x_t)}{\|\nabla f_t(\x_t)\|}\right\|\leq\delta\leq 1.
\end{align*}
\end{lemma}

The proof of \lem{Comparison-AdaNGD-convergence} is deferred to \append{NGD}. We now prove \thm{CCO} using \lem{Comparison-AdaNGD-convergence}.

\begin{proof}[Proof of \thm{CCO}]
We show that \algo{Comparison-AdaNGD} solves \prob{CCO} by contradiction. Assume that the output of \algo{Comparison-AdaNGD} is not an $\epsilon$-optimal point of $f$, or equivalently, $f(\x_t)-f^*\geq\epsilon$ for any $t\in[T]$. 
This leads to
\begin{align*}
\|\nabla f(\x_t)\|\geq\frac{f(\x_t)-f^*}{\|\x_t-\x^*\|}\geq\frac{\epsilon}{2R},\quad\forall t\in[T]
\end{align*}
given that $f$ is convex. Hence, \thm{Comparison-GDE} promises that
\begin{align*}
\left\|\hat{\g}_t-\frac{\nabla f(\x_t)}{\|\nabla f(\x_t)\|}\right\|\leq\delta\leq 1.
\end{align*}
With these approximate gradient directions, by \lem{Comparison-AdaNGD-convergence} we can derive that
\begin{align*}
\min_{t\in[T]}f(\x_t)-f^*
&\leq2L(2R\sqrt{2T}+2T\delta R)^2/T^2\leq\epsilon,
\end{align*}
contradiction. This proves the correctness of \algo{Comparison-AdaNGD}.
The query complexity of \algo{Comparison-AdaNGD} only comes from the gradient direction estimation step in \lin{AdaNGD-GDE}, which equals
\begin{align*}
T\cdot O(n\log(n/\delta))=O\left(\frac{nLR^2}{\epsilon}\log\left(\frac{nLR^2}{\epsilon}\right)\right).
\end{align*}
\end{proof}

\subsection{Comparison-based cutting plane method}\label{sec:cutting}
In this subsection, we provide a comparison-based cutting plane method that solves \prob{CCO}. We begin by introducing the basic notation and concepts of cutting plane methods, which are algorithms that solves the feasibility problem defined as follows.

\begin{problem}[Feasibility Problem, \cite{jiang2020improved,sidford2023quantum}]\label{prob:feasibility}
We are given query access to a separation oracle for a set $K\subset\R^n$ such that on query $\x\in\R^n$ the oracle outputs a vector $\vect{c}$ and either $\vect{c} = \0$, in which case $\x \in K$, or $\vect{c} \neq \0$, in which case $H\coloneqq\{\vect{z}\colon\vect{c}^\top\vect{z}\leq\vect{c}^\top \x\}\supset K$. The goal is to query a point $\x\in K$.
\end{problem}

\cite{jiang2020improved} developed a cutting plane method that solves \prob{feasibility} using $O(n\log(nR/r))$ queries to a separation oracle where $R$ and $r$ are parameters related to the convex set $\mathcal{K}$.

\begin{lemma}[Theorem 1.1, \cite{jiang2020improved}]\label{lem:classical-cutting-plane}
There is a cutting plane method which solves \prob{feasibility} using at most $C\cdot n\log(nR/r)$ queries for some constant $C$, given that the set $K$ is contained in the ball of radius $R$ centered at the origin and it contains a ball of radius $r$. 
\end{lemma}

Refs.~\cite{nemirovski1994efficient,lee2015faster} showed that, running cutting plane method on a Lipschitz convex function $f$ with the separation oracle being the gradient of $f$ would yield a sequence of points where at least one of them is $\epsilon$-optimal. Furthermore, Ref.~\cite{sidford2023quantum} showed that even if we cannot access the exact gradient value of $f$, it suffices to use an approximate gradient estimate with absolute error at most $O(\epsilon/R)$. 

In this work, we show that this result can be extended to the case where we have an estimate of the gradient direction instead of the gradient itself. Specifically, we prove the following result.

\begin{theorem}\label{thm:cutting-plane}
There exists an algorithm based on cutting plane method that solves \prob{CCO} using $O(n^2\log\small(nLR^2/\epsilon\small))$ queries.
\end{theorem}

Note that \thm{cutting-plane} improves the prior state-of-the-art  from $\tilde{O}(n^{4})$ by~\cite{karabag2021smooth} to $\tilde{O}(n^{2})$. 

\begin{proof}[Proof of \thm{cutting-plane}]
The proof follows a similar intuition as the proof of Proposition 1 in \cite{sidford2023quantum}.
Define $\mathcal{K}_{\epsilon/2}$ to be the set of $\epsilon/2$-optimal points of $f$, and $\mathcal{K}_{\epsilon}$ to be the set of $\epsilon$-optimal points of $f$. Given that $f$ is $L$-smooth, $\mathcal{K}_{\epsilon/2}$ must contain a ball of radius at least $r_\mathcal{K} = \sqrt{\epsilon/L}$ since for any $\x$ with $\|\x-\x^*\|\leq r_\mathcal{K}$ we have
\begin{align*}
f(\x)-f(\x^*)\leq L\|\x-\x^*\|^2/2\leq\epsilon/2.
\end{align*}
We apply the cutting plane method, as described in \lem{classical-cutting-plane}, to query a point in $\mathcal{K}_{\epsilon/2}$, which is a subset of the ball $\mathbb{B}_{2R}(\0)$. To achieve this, at each query $\x$ of the cutting plane method, we use \texttt{Comparison-GDE}$(\x,\delta,\gamma)$, our comparison-based gradient direction estimation algorithm (\algo{Comparison-GDE}), as the separation oracle for the cutting plane method, where we set
\begin{align*}
\delta=\frac{1}{16R}\sqrt{\frac{\epsilon}{L}},\qquad\gamma=\sqrt{2L\epsilon}.
\end{align*}

We show that any query outside of $\mathcal{K}_{\epsilon}$ to \texttt{Comparison-GDE}$(\x,\delta,\gamma)$ will be a valid separation oracle for $\mathcal{K}_{\epsilon/2}$. In particular, if we ever queried \texttt{Comparison-GDE}$(\x,\delta,\gamma)$ at any $\x\in\B_{2R}(\0)\setminus \mathcal{K}_{\epsilon}$ with output being $\hat{\g}$, for any $\y\in \mathcal{K}_{\epsilon/2}$ we have
\begin{align*}
\left<\hat{\g},\y-\x\right>&\leq\left\<\frac{\nabla f(\x)}{\|\nabla f(\x)\|},\y-\x\right\>+\left\|\hat{\g}-\frac{\nabla f(\x)}{\|\nabla f(\x)\|}\right\|\cdot\|\y-\x\|\\
&\leq \frac{f(\y)-f(\x)}{\|\nabla f(\x)\|}+\left\|\hat{\g}-\frac{\nabla f(\x)}{\|\nabla f(\x)\|}\right\|\cdot\|\y-\x\|\leq -\frac{\epsilon}{2}+\frac{\epsilon}{10R}\cdot 4R<0,
\end{align*}
where
\begin{align*}
\|\nabla f(\x)\|\geq (f(\x)-f^*)/\|\x-\x^*\|\geq(f(\x)-f^*)/(2R)
\end{align*}
given that $f$ is convex. Combined with \thm{Comparison-GDE}, it guarantees that
\begin{align*}
\left\|\hat{\g}-\frac{\nabla f(\x)}{\|\nabla f(\x)\|}\right\|\leq\delta=\frac{1}{16R}\sqrt{\frac{\epsilon}{L}}.
\end{align*}
Hence,
\begin{align*}
\left<\hat{\g},\y-\x\right>
&\leq \frac{f(\y)-f(\x)}{\|\nabla f(\x)\|}+\left\|\hat{\g}-\frac{\nabla f(\x)}{\|\nabla f(\x)\|}\right\|\cdot\|\y-\x\|\leq -\frac{1}{2}\sqrt{\frac{\epsilon}{2L}}+\frac{1}{16R}\sqrt{\frac{\epsilon}{L}}\cdot 4R<0,
\end{align*}
indicating that $\hat{\g}$ is a valid separation oracle for the set $\mathcal{K}_{\epsilon/2}$. Consequently, by \lem{classical-cutting-plane}, after $Cn\log(nR/r_\mathcal{K})$ iterations, at least one of the queries must lie within $\mathcal{K}_{\epsilon}$, and we can choose the query with minimum function value to output, which can be done by making $Cn\log(nR/r_\mathcal{K})$ comparisons.

Note that in each iteration $O(n\log (n/\delta))$ queries to $O_{f}^{\comp}$ \eq{comparison} are needed. Hence, the overall query complexity equals
\begin{align*}
&Cn\log(nR/r_\mathcal{K})\cdot O(n\log (n/\delta))+Cn\log(nR/r_\mathcal{K})=O\left(n^2\log\left(nLR^2/\epsilon\right)\right).
\end{align*}
\end{proof}



\section{Nonconvex Optimization by Comparisons}

In this section, we study nonconvex optimization with function value comparisons. We first develop an algorithm that finds an $\epsilon$-FOSP of a smooth nonconvex function in \sec{FOSP}. Then in \sec{SOSP}, we further develop an algorithm that finds an $\epsilon$-SOSP of a nonconvex function that is smooth and Hessian-Lipschitz.

\subsection{First-order stationary point computation by comparisons}\label{sec:FOSP}

In this subsection, we focus on the problem of finding an $\epsilon$-FOSP of a smooth nonconvex function by making function value comparisons.

\begin{problem}[Comparison-based first-order stationary point computation]\label{prob:Comparison-FOSP}
In the Comparison-based first-order stationary point computation (Comparison-FOSP) problem we are given query access to a comparison oracle $O_{f}^{\comp}$ \eq{comparison} for an $L$-smooth (possibly) nonconvex function $f\colon\R^n\to\R$ satisfying $f(\0)-\inf_{\x}f(\x)\leq\Delta$. The goal is to output an $\epsilon$-FOSP of $f$. 
\end{problem}

We develop a comparison-based normalized gradient descent algorithm that solves \prob{Comparison-FOSP}.

\begin{algorithm2e}[t]
	\caption{Comparison-based Approximate Normalized Gradient Descent (Comparison-NGD)}
	\label{algo:Comparison-NGD}
	\LinesNumbered
	\DontPrintSemicolon
    \KwInput{Function $f\colon\R^n\to\R$, $\Delta$, precision $\epsilon$}
    $T\leftarrow \frac{18L\Delta}{\epsilon^2}$, $\x_0\leftarrow\0$\;
    \For{$t=0,\ldots,T-1$}{
        $\hat{\g}_t\leftarrow$\texttt{Comparison-GDE}$(\x_t,1/6,\epsilon/12)$ \label{lin:NGD-GDE}
        \;
        $\x_t=\x_{t-1}-\epsilon\hat{\g}_t/(3L)$\;
    }
    Uniformly randomly select $\x_\out$ from $\{\x_0,\ldots,\x_T\}$\;
    \Return $\x_\out$\;
\end{algorithm2e}

\begin{theorem}\label{thm:Comparison-FOSP}
With success probability at least $2/3$, \algo{Comparison-NGD} solves \prob{Comparison-FOSP} using $O(L\Delta n\log n/\epsilon^2)$ queries.
\end{theorem}
The proof of \thm{Comparison-FOSP} is deferred to \append{FOSP-proof}.



\subsection{Escaping saddle points of nonconvex functions by comparisons}\label{sec:SOSP}

In this subsection, we focus on the problem of escaping from saddle points, i.e., finding an $\epsilon$-SOSP of a nonconvex function that is smooth and Hessian-Lipschitz, by making function value comparisons. 

\begin{problem}[Comparison-based escaping from saddle point]\label{prob:Comparison-SOSP}
In the Comparison-based escaping from saddle point (Comparison-SOSP) problem we are given query access to a comparison oracle $O_{f}^{\comp}$ \eq{comparison} for a (possibly) nonconvex function $f\colon\R^n\to\R$ satisfying $f(\0)-\inf_{\x}f(\x)\leq\Delta$ that is $L$-smooth and $\rho$-Hessian Lipschitz. The goal is to output an $\epsilon$-SOSP of $f$. 
\end{problem}

Our algorithm for \prob{Comparison-SOSP} given in \algo{Comparison-PNGD} is a combination of comparison-based normalized gradient descent and comparison-based negative curvature descent (\texttt{Comparison-NCD}). Specifically, \texttt{Comparison-NCD} is built upon our comparison-based negative curvature finding algorithms, \texttt{Comparison-NCF1} (\algo{Comparison-NCF1}) and \texttt{Comparison-NCF2} (\algo{Comparison-NCF2}) that work when the gradient is small or large respectively, and can decrease the function value efficiently when applied at a point with a large negative curvature.

\begin{algorithm2e}[ht]
	\caption{Comparison-based Perturbed Normalized Gradient Descent (Comparison-PNGD)}
	\label{algo:Comparison-PNGD}
	\LinesNumbered
	\DontPrintSemicolon
    \KwInput{Function $f\colon\R^n\to\R$, $\Delta$, precision $\epsilon$}
    $\mathcal{S}\leftarrow 350\Delta\sqrt{\frac{\rho}{\epsilon^3}}$, $\delta\leftarrow\frac{1}{6}$, $\x_{1,0}\leftarrow\0$\;
    $\mathscr{T}\leftarrow\frac{384L^2\sqrt{n}}{\delta\rho\epsilon}\log\frac{36nL}{\sqrt{\rho\epsilon}}$, $p\leftarrow\frac{100}{\T}\log\mathcal{S}$\;
    \For{$s=1,\ldots,\mathcal{S}$}{
        \For{$t=0,\ldots,\mathscr{T}-1$}{
        $\hat{\g}_t\leftarrow$\texttt{Comparison-GDE}$(\x_{s,t},\delta,\gamma)$\label{lin:PNGD-GDE}\;
        $\y_{s,t}\leftarrow\x_{s,t}-\epsilon\hat{\g}_t/(3L)$\;
        Choose $\x_{s,t+1}$ to be the point between $\{x_{s,t},\y_{s,t}\}$ with smaller function value\;
        $\x_{s,t+1}'\leftarrow
        \begin{cases}
        \0,\text{ w.p. }1-p\\
        \texttt{Comparison-NCD}(\x_{s,t+1},\epsilon,\delta),\text{ w.p. }p
        \end{cases}$\label{lin:PNGD-NCD}
        }
        Choose $\x_{s+1,0}$ among $\{\x_{s,0},\ldots,\x_{s,\T},\x_{s,0}',\ldots,\x_{s,\T}'\}$ with the smallest function value.\;
        $\x_{s+1,0}'\leftarrow\begin{cases}
        \0,\text{ w.p. }1-p\\
        \texttt{Comparison-NCD}(\x_{s+1,0},\epsilon,\delta),\text{ w.p. }p
        \end{cases}$
    }
    Uniformly randomly select $s_{\mathrm{out}}\in\{1,\ldots,\mathcal{S}\}$ and $t_{\mathrm{out}}\in[\T]$\;
    \Return $\x_{s_{\mathrm{out}},t_{\mathrm{out}}}$\;
\end{algorithm2e}

\begin{algorithm2e}[!h]
	\caption{Comparison-based Negative Curvature Descent (Comparison-NCD)}
	\label{algo:Comparison-NCD}
	\LinesNumbered
	\DontPrintSemicolon
    \KwInput{Function $f\colon\R^n\to\R$, precision $\epsilon$, input point $\z$, error probability $\delta$}
    $\v_1\leftarrow$\texttt{Comparison-NCF1}$(\z,\epsilon,\delta)$\;
    $\v_2\leftarrow$\texttt{Comparison-NCF2}$(\z,\epsilon,\delta)$\;
    $\z_{1,+}=\z+\frac{1}{2}\sqrt{\frac{\epsilon}{\rho}}\v_1$, $\z_{1,-}=\z-\frac{1}{2}\sqrt{\frac{\epsilon}{\rho}}\v_1$, $\z_{2,+}=\z+\frac{1}{2}\sqrt{\frac{\epsilon}{\rho}}\v_2$, $\z_{2,-}=\z-\frac{1}{2}\sqrt{\frac{\epsilon}{\rho}}\v_2$\;
    \Return $\z_{\out}\in\{\z_{1,+},\z_{1,-},\z_{2,+},\z_{2,-}\}$ with the smallest function value.
\end{algorithm2e}

\begin{lemma}\label{lem:NCD}
In the setting of \prob{Comparison-SOSP}, for any $\z$ satisfying $\lambda_{\min}(\nabla^2f(\x))\leq-\sqrt{\rho\epsilon}$, \algo{Comparison-NCD} outputs a point $\z_{\out}\in\R^n$ satisfying 
\begin{align*}
f(\z_\out)-f(\z)\leq-\frac{1}{48}\sqrt{\frac{\epsilon^3}{\rho}}
\end{align*}
with success probability at least $1-\zeta$ using $O\big(\frac{L^2n^{3/2}}{\zeta\rho\epsilon}\log^2\frac{nL}{\zeta\sqrt{\rho\epsilon}}\big)$ queries.
\end{lemma}

The proof of \lem{NCD} is deferred to \append{NCD-proof}. Next, we present the main result of this subsection, which describes the complexity of solving \prob{Comparison-SOSP} using \algo{Comparison-PNGD}.

\begin{theorem}\label{thm:Comparison-SOSP}
With success probability at least $2/3$, \algo{Comparison-PNGD} solves \prob{Comparison-SOSP} using an expected $O\big(\frac{\Delta L^2n^{3/2}}{\rho^{1/2}\epsilon^{5/2}}\log^3\frac{nL}{\sqrt{\rho\epsilon}}\big)$ queries.
\end{theorem}
The proof of \thm{Comparison-SOSP} is deferred to \append{saddle-proof}.
